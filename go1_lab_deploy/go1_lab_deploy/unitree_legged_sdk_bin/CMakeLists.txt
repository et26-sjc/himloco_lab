cmake_minimum_required(VERSION 3.10)
project(himloco_go1_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)

# The gate can be tested on a host with no robot SDK or LCM installation.
option(HIM_GUARD_TEST_ONLY "Build only SDK-independent guard tests" OFF)
add_executable(bridge_guard_test ../tests/himloco/bridge_guard_test.cpp)
target_link_libraries(bridge_guard_test PRIVATE Threads::Threads)
enable_testing()
add_test(NAME bridge_guard_test COMMAND bridge_guard_test)
if(HIM_GUARD_TEST_ONLY)
    return()
endif()

if(NOT UNIX)
    message(FATAL_ERROR "The real Unitree SDK bridge must be built on target Linux")
endif()
set(UNITREE_SDK_ROOT "" CACHE PATH "Installed Unitree Legged SDK Go1/v3.8 checkout")
if(NOT EXISTS "${UNITREE_SDK_ROOT}/include/unitree_legged_sdk/unitree_legged_sdk.h")
    message(FATAL_ERROR "Set UNITREE_SDK_ROOT explicitly to the target SDK checkout")
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)")
    set(UNITREE_ARCH arm64)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)")
    set(UNITREE_ARCH amd64)
else()
    message(FATAL_ERROR "Unsupported Unitree SDK architecture: ${CMAKE_SYSTEM_PROCESSOR}")
endif()
find_library(UNITREE_SDK_LIBRARY NAMES unitree_legged_sdk
    PATHS "${UNITREE_SDK_ROOT}/lib/cpp/${UNITREE_ARCH}" NO_DEFAULT_PATH)
if(NOT UNITREE_SDK_LIBRARY)
    message(FATAL_ERROR "Unitree SDK library not found for ${UNITREE_ARCH}")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(LCM REQUIRED IMPORTED_TARGET lcm)
find_package(Boost REQUIRED)
find_program(LCM_GEN_EXECUTABLE lcm-gen)
if(NOT LCM_GEN_EXECUTABLE)
    message(FATAL_ERROR "Install lcm-gen; stale pre-generated headers are not used")
endif()

set(LCM_NAMES state_estimator_lcmt leg_control_data_lcmt pd_tau_targets_lcmt rc_command_lcmt)
set(GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GENERATED_DIR}")
set(SOURCE_DIGESTS "")
foreach(SOURCE_FILE lcm_position.cpp bridge_guard.hpp)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${SOURCE_FILE}")
    file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${SOURCE_FILE}" DIGEST)
    string(APPEND SOURCE_DIGESTS "${DIGEST}")
endforeach()
foreach(TYPE_NAME IN LISTS LCM_NAMES)
    set(SCHEMA "${CMAKE_CURRENT_SOURCE_DIR}/../lcm_types/${TYPE_NAME}.lcm")
    list(APPEND SCHEMAS "${SCHEMA}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SCHEMA}")
    list(APPEND GENERATED_HEADERS "${GENERATED_DIR}/${TYPE_NAME}.hpp")
    file(SHA256 "${SCHEMA}" DIGEST)
    string(APPEND SOURCE_DIGESTS "${DIGEST}")
endforeach()
string(SHA256 HIM_SOURCE_HASH "${SOURCE_DIGESTS}")
add_custom_command(OUTPUT ${GENERATED_HEADERS}
    COMMAND "${LCM_GEN_EXECUTABLE}" --cpp --cpp-hpath "${GENERATED_DIR}" ${SCHEMAS}
    DEPENDS ${SCHEMAS} VERBATIM)
add_executable(lcm_position_him lcm_position.cpp ${GENERATED_HEADERS})
target_include_directories(lcm_position_him PRIVATE
    "${UNITREE_SDK_ROOT}/include" "${GENERATED_DIR}" ${Boost_INCLUDE_DIRS})
target_compile_definitions(lcm_position_him PRIVATE HIM_BUILD_SOURCE_SHA256="${HIM_SOURCE_HASH}")
target_link_libraries(lcm_position_him PRIVATE "${UNITREE_SDK_LIBRARY}" PkgConfig::LCM Threads::Threads)
set_target_properties(lcm_position_him PROPERTIES BUILD_RPATH "${UNITREE_SDK_ROOT}/lib/cpp/${UNITREE_ARCH}")
message(STATUS "HIM bridge source SHA256: ${HIM_SOURCE_HASH}")
